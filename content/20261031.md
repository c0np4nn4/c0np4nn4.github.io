var HASH = "96061e92f58e4bdcdee73df36183fe3ac64747c81c26f6c83aada8d2aabb1864";

async function sha256(s) {
  var buf = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(s));
  return Array.from(new Uint8Array(buf), function (b) {
    return b.toString(16).padStart(2, "0");
  }).join("");
}

document.getElementById("pw").addEventListener("input", async function (e) {
  document.getElementById("secret").hidden = (await sha256(e.target.value)) !== HASH;
});
